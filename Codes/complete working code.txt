// =====================================================
// ESTIMATED DEMO pH
// =====================================================

float estimatedPH(int moisturePercent)
{
  if (moisturePercent <= 20)
    return 6.4;

  else if (moisturePercent <= 40)
    return 6.5;

  else if (moisturePercent <= 60)
    return 6.7;

  else if (moisturePercent <= 80)
    return 6.8;

  else
    return 6.9;
}


// =====================================================
// WATER LEVEL STATUS
// =====================================================

String getWaterLevel(int value)
{
  if (value < WATER_EMPTY)
    return "EMPTY";

  else if (value < WATER_LOW)
    return "LOW";

  else if (value < WATER_MEDIUM)
    return "MEDIUM";

  else
    return "HIGH";
}


// =====================================================
// AVERAGE SOIL SENSOR
// =====================================================

int readSoilSensor()
{
  long total = 0;

  for (int i = 0; i < 10; i++)
  {
    total += analogRead(SOIL_PIN);
    delay(5);
  }

  return total / 10;
}


// =====================================================
// AVERAGE WATER SENSOR
// =====================================================

int readWaterSensor()
{
  long total = 0;

  for (int i = 0; i < 10; i++)
  {
    total += analogRead(WATER_LEVEL_PIN);
    delay(5);
  }

  return total / 10;
}


// =====================================================
// PUMP CONTROL
// ACTIVE LOW RELAY
// =====================================================

void setPump(bool state)
{
  // Detect OFF -> ON transition
  if (state && !pumpState)
  {
    pumpStartTime = millis();
  }

  pumpState = state;

  if (pumpState)
  {
    digitalWrite(RELAY_PIN, LOW);
  }
  else
  {
    digitalWrite(RELAY_PIN, HIGH);
  }
}


// =====================================================
// SENSOR READING + AUTOMATIC WATERING
// =====================================================

void updateSensors()
{
  // -------------------------------------
  // SOIL MOISTURE
  // -------------------------------------

  soilRaw = readSoilSensor();

  moisture = map(
    soilRaw,
    SOIL_DRY,
    SOIL_WET,
    0,
    100
  );

  moisture = constrain(moisture, 0, 100);


  // -------------------------------------
  // ESTIMATED pH
  // -------------------------------------

  estimatedPhValue = estimatedPH(moisture);


  // -------------------------------------
  // DHT11
  // -------------------------------------

  float newTemperature = dht.readTemperature();
  float newHumidity = dht.readHumidity();

  if (!isnan(newTemperature) &&
      !isnan(newHumidity))
  {
    temperature = newTemperature;
    humidity = newHumidity;

    dhtOK = true;
  }
  else
  {
    dhtOK = false;

    Serial.println("WARNING: DHT11 reading failed");
  }


  // -------------------------------------
  // WATER LEVEL
  // -------------------------------------

  waterRaw = readWaterSensor();

  tankStatus = getWaterLevel(waterRaw);


  // -------------------------------------
  // AUTOMATIC PUMP CONTROL
  // -------------------------------------

  // No tank water
  if (waterRaw < WATER_EMPTY)
  {
    setPump(false);
  }

  else
  {
    // Dry soil
    if (moisture < PUMP_ON_THRESHOLD &&
        !pumpState)
    {
      setPump(true);
    }

    // Soil sufficiently wet
    else if (moisture >= PUMP_OFF_THRESHOLD)
    {
      setPump(false);
    }
  }


  // -------------------------------------
  // SERIAL MONITOR
  // -------------------------------------

  Serial.print("Soil Raw: ");
  Serial.print(soilRaw);

  Serial.print(" | Moisture: ");
  Serial.print(moisture);
  Serial.print("%");

  Serial.print(" | pH(est): ");
  Serial.print(estimatedPhValue, 1);

  if (dhtOK)
  {
    Serial.print(" | Temp: ");
    Serial.print(temperature, 1);
    Serial.print("C");

    Serial.print(" | Humidity: ");
    Serial.print(humidity, 0);
    Serial.print("%");
  }
  else
  {
    Serial.print(" | DHT: ERROR");
  }

  Serial.print(" | Water Raw: ");
  Serial.print(waterRaw);

  Serial.print(" | Tank: ");
  Serial.print(tankStatus);

  Serial.print(" | Pump: ");

  if (pumpState)
    Serial.println("ON");
  else
    Serial.println("OFF");
}

// =====================================================
// NORMAL LCD PAGES
// =====================================================

void updateLCD()
{
  // Keep showing IP until browser opens dashboard
  if (!dashboardConnected)
  {
    return;
  }

  if (!lcdImmediateRefresh &&
      millis() - lastLCDChange < LCD_INTERVAL)
  {
    return;
  }

  lcdImmediateRefresh = false;

  lastLCDChange = millis();

  lcd.clear();


  // ===================================================
  // PAGE 1
  // ===================================================

  if (lcdPage == 0)
  {
    lcd.setCursor(0, 0);

    lcd.print("Moist:");
    lcd.print(moisture);
    lcd.print("%");

    lcd.setCursor(0, 1);

    lcd.print("pH(est):");
    lcd.print(estimatedPhValue, 1);
  }


  // ===================================================
  // PAGE 2
  // ===================================================

  else if (lcdPage == 1)
  {
    if (dhtOK)
    {
      lcd.setCursor(0, 0);

      lcd.print("Temp:");
      lcd.print(temperature, 1);
      lcd.print("C");

      lcd.setCursor(0, 1);

      lcd.print("Hum:");
      lcd.print(humidity, 0);
      lcd.print("%");
    }

    else
    {
      lcd.setCursor(0, 0);
      lcd.print("DHT11 ERROR");

      lcd.setCursor(0, 1);
      lcd.print("Check Sensor");
    }
  }


  // ===================================================
  // PAGE 3
  // ===================================================

  else if (lcdPage == 2)
  {
    lcd.setCursor(0, 0);

    lcd.print("Tank:");
    lcd.print(tankStatus);

    lcd.setCursor(0, 1);

    lcd.print("Pump:");

    if (pumpState)
      lcd.print("ON");
    else
      lcd.print("OFF");
  }


  // ===================================================
  // PAGE 4
  // ===================================================

  else
  {
    lcd.setCursor(0, 0);

    if (moisture <= 20)
      lcd.print("Soil:VERY DRY");

    else if (moisture <= 40)
      lcd.print("Soil:DRY");

    else if (moisture <= 60)
      lcd.print("Soil:NORMAL");

    else if (moisture <= 80)
      lcd.print("Soil:MOIST");

    else
      lcd.print("Soil:VERY WET");


    lcd.setCursor(0, 1);

    if (waterRaw < WATER_EMPTY)
      lcd.print("NO WATER!");

    else
      lcd.print("Auto Mode");
  }


  lcdPage++;

  if (lcdPage > 3)
    lcdPage = 0;
}

      data.uptime % 60;


    document.getElementById(
      "uptime"
    ).textContent =
      minutes + "m " +
      seconds + "s";


    /* HISTORY */

    moistureHistory.push(
      data.moisture
    );

    if(moistureHistory.length > 40)
      moistureHistory.shift();


    drawChart();

  }

  catch(error){

    document.getElementById(
      "systemStatus"
    ).textContent =
      "ESP32 connection lost";

  }

}


/* ==========================================
   SIMPLE LIVE GRAPH
   ========================================== */

function drawChart(){

  const canvas =
    document.getElementById(
      "chart"
    );

  const rect =
    canvas.getBoundingClientRect();

  const dpr =
    window.devicePixelRatio || 1;


  canvas.width =
    rect.width * dpr;

  canvas.height =
    rect.height * dpr;


  const ctx =
    canvas.getContext("2d");

  ctx.scale(
    dpr,
    dpr
  );


  const w =
    rect.width;

  const h =
    rect.height;


  ctx.clearRect(
    0,
    0,
    w,
    h
  );


  /* GRID */

  ctx.strokeStyle =
    "rgba(255,255,255,.06)";

  ctx.lineWidth = 1;


  for(let i=0;i<=4;i++){

    const y =
      (h / 4) * i;

    ctx.beginPath();

    ctx.moveTo(
      0,
      y
    );

    ctx.lineTo(
      w,
      y
    );

    ctx.stroke();

  }


  if(moistureHistory.length < 2)
    return;


  /* LINE */

  ctx.strokeStyle =
    "#4de4bd";

  ctx.lineWidth = 2.5;

  ctx.beginPath();


  moistureHistory.forEach(
    (value,index)=>{

      const x =
        index /
        (moistureHistory.length - 1)
        * w;

      const y =
        h -
        (value / 100 * h);


      if(index === 0)
        ctx.moveTo(x,y);

      else
        ctx.lineTo(x,y);

    }
  );


  ctx.stroke();


  /* GLOW */

  ctx.shadowColor =
    "#4de4bd";

  ctx.shadowBlur = 8;

}

  drawChart

  else
    json += "null";

  json += ",";

  json += "\"humidity\":";

  if (dhtOK)
    json += String(humidity, 0);
  else
    json += "null";

  json += ",";

  json += "\"waterRaw\":";
  json += String(waterRaw);
  json += ",";

  json += "\"tank\":\"";
  json += tankStatus;
  json += "\",";

  json += "\"pump\":";
  json += pumpState ? "true" : "false";
  json += ",";

  json += "\"ip\":\"";
  json += WiFi.softAPIP().toString();

    "Cache-Control",
    "no-store"
  );

  server.send(
    200,
    "application/json",
    json
  );

void setupWebServer()

  server.on(

// =====================================================
// SETUP
// =====================================================

void setup()
{
  Serial.begin(115200);


  // ===================================================
  // LCD
  // ===================================================

  Wire.begin(21, 22);

  lcd.init();

  lcd.backlight();


  // ===================================================
  // DHT
  // ===================================================

  dht.begin();


  // ===================================================
  // RELAY
  // ===================================================

  pinMode(
    RELAY_PIN,
    OUTPUT
  );

  setPump(false);


  // ===================================================
  // START SCREEN
  // ===================================================

  lcd.clear();

  lcd.setCursor(0, 0);

  lcd.print(
    "Smart Watering"
  );

  lcd.setCursor(0, 1);

  lcd.print(
    "Starting WiFi..."
  );

  delay(1500);


  // ===================================================
  // START ESP32 ACCESS POINT
  // ===================================================

  WiFi.mode(
    WIFI_AP
  );


  WiFi.softAPConfig(
    localIP,
    gateway,
    subnet
  );


  bool apStarted =
    WiFi.softAP(
      AP_SSID,
      AP_PASSWORD
    );


  if (!apStarted)
  {
    lcd.clear();

    lcd.setCursor(0, 0);

    lcd.print(
      "WiFi AP ERROR"
    );

    Serial.println(
      "Failed to start WiFi Access Point"
    );
  }

  else
  {
    Serial.println();

    Serial.println(
      "================================="
    );

    Serial.println(
      "SMART WATERING DASHBOARD"
    );

    Serial.println(
      "================================="
    );

    Serial.print(
      "WiFi: "
    );

    Serial.println(
      AP_SSID
    );

    Serial.print(
      "Password: "
    );

    Serial.println(
      AP_PASSWORD
    );

    Serial.print(
      "Dashboard: http://"
    );

    Serial.println(
      WiFi.softAPIP()
    );

    Serial.println(
      "================================="
    );


    // Show IP until browser connects
    showDashboardIP();
  }


  // ===================================================
  // WEB SERVER
  // ===================================================

  setupWebServer();


  // Make first sensor update immediate
  lastSensorRead =
    millis() - SENSOR_INTERVAL;
}


// =====================================================
// MAIN LOOP
// =====================================================

void loop()
{
  // Keep web server responsive
  server.handleClient();


  // ===================================================
  // SENSOR UPDATE
  // ===================================================

  if (
    millis() - lastSensorRead
    >= SENSOR_INTERVAL
  )
  {
    lastSensorRead =
      millis();

    updateSensors();
  }


  // ===================================================
  // PUMP SAFETY
  // ===================================================

  if (pumpState)
  {
    if (
      millis() - pumpStartTime
      >= MAX_PUMP_RUNTIME
    )
    {
      setPump(false);

      Serial.println(
        "SAFETY: Pump maximum runtime reached"
      );
    }
  }


  // ===================================================
  // LCD
  // ===================================================

  updateLCD();


  delay(2);
}
